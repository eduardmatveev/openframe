cmake_minimum_required(VERSION 3.24)
project(OpenFrame VERSION 0.4.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)
option(OPENFRAME_BUILD_TESTS "Build synthetic-fixture tests" ON)
option(OPENFRAME_BUILD_UI "Build desktop application" ON)
set(OPENFRAME_FFMPEG_ROOT "" CACHE PATH "FFmpeg installation prefix (include/lib/bin)")
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Concurrent Gui Widgets Network Multimedia Test)
if(NOT OPENFRAME_LAST_FFMPEG_ROOT STREQUAL OPENFRAME_FFMPEG_ROOT)
    foreach(component avformat avcodec avutil swscale swresample avfilter)
        unset(${component}_INCLUDE_DIR CACHE)
        unset(${component}_LIBRARY CACHE)
    endforeach()
    set(OPENFRAME_LAST_FFMPEG_ROOT "${OPENFRAME_FFMPEG_ROOT}" CACHE INTERNAL "SDK used for cached FFmpeg paths" FORCE)
endif()
foreach(component avformat avcodec avutil swscale swresample avfilter)
    find_path(${component}_INCLUDE_DIR lib${component}/${component}.h HINTS "${OPENFRAME_FFMPEG_ROOT}/include" REQUIRED)
    find_library(${component}_LIBRARY NAMES ${component} lib${component} HINTS "${OPENFRAME_FFMPEG_ROOT}/lib" "${OPENFRAME_FFMPEG_ROOT}/bin" REQUIRED)
    add_library(FFmpeg::${component} UNKNOWN IMPORTED)
    set_target_properties(FFmpeg::${component} PROPERTIES IMPORTED_LOCATION "${${component}_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${${component}_INCLUDE_DIR}")
endforeach()
add_library(openframe_core STATIC
    src/core/Project.cpp src/core/Project.h
    src/core/ReviewSession.cpp src/core/ReviewSession.h
    src/core/ViewerGeometry.cpp src/core/ViewerGeometry.h
    src/media/AnnotationRenderer.cpp src/media/AnnotationRenderer.h
    src/media/MediaDecoder.cpp src/media/MediaDecoder.h
    src/media/ColorPipeline.cpp src/media/ColorPipeline.h
    src/media/SequenceReader.cpp src/media/SequenceReader.h
    src/media/AudioTimeStretcher.cpp src/media/AudioTimeStretcher.h
    src/media/AudioOutputConverter.cpp src/media/AudioOutputConverter.h
    src/media/FrameComposer.cpp src/media/FrameComposer.h
    src/media/Exporter.cpp src/media/Exporter.h
    src/integration/LocalPathPolicy.cpp src/integration/LocalPathPolicy.h
)
target_include_directories(openframe_core PUBLIC src)
target_link_libraries(openframe_core PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Multimedia PRIVATE Qt6::Concurrent FFmpeg::avformat FFmpeg::avcodec FFmpeg::avutil FFmpeg::swscale FFmpeg::swresample FFmpeg::avfilter)
if(WIN32)
    target_link_libraries(openframe_core PRIVATE advapi32)
endif()
add_library(openframe_integration STATIC src/integration/ApiServer.cpp src/integration/ApiServer.h)
target_link_libraries(openframe_integration PUBLIC openframe_core Qt6::Network)
set(OPENFRAME_TARGETS openframe_core openframe_integration)
if(OPENFRAME_BUILD_UI)
    file(GLOB OPENFRAME_UI_SOURCES CONFIGURE_DEPENDS src/ui/*.cpp src/ui/*.h src/ui/*.qrc)
    add_executable(OpenFrame WIN32 src/main.cpp ${OPENFRAME_UI_SOURCES})
    target_link_libraries(OpenFrame PRIVATE openframe_integration)
    if(WIN32)
        enable_language(RC)
        set(icon_dir "${CMAKE_CURRENT_SOURCE_DIR}/resources/icons")
        set_source_files_properties("${icon_dir}/OpenFrame.rc" PROPERTIES
            INCLUDE_DIRECTORIES "${icon_dir}"
            OBJECT_DEPENDS "${icon_dir}/openframe.ico")
        target_sources(OpenFrame PRIVATE "${icon_dir}/OpenFrame.rc")
        target_link_libraries(OpenFrame PRIVATE Qt6::GuiPrivate shell32 ole32)
    endif()
    list(APPEND OPENFRAME_TARGETS OpenFrame)
    install(TARGETS OpenFrame RUNTIME DESTINATION .)
endif()
if(MSVC)
    foreach(target ${OPENFRAME_TARGETS})
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8;/permissive-;/EHsc>")
        target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    endforeach()
endif()
if(OPENFRAME_BUILD_TESTS)
    target_compile_definitions(openframe_core PRIVATE OPENFRAME_TESTING)
    enable_testing()
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME openframe_maya_adapter COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration" -p test_maya.py -v)
        if(WIN32)
            add_test(NAME openframe_ffmpeg_policy COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/verify_ffmpeg.py" "${OPENFRAME_FFMPEG_ROOT}/bin" --output "${CMAKE_BINARY_DIR}/ffmpeg-policy.json")
        endif()
    endif()
    add_subdirectory(tests/core)
    if(OPENFRAME_BUILD_UI AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/ui/UiTests.cpp")
        add_executable(openframe_ui_tests tests/ui/UiTests.cpp ${OPENFRAME_UI_SOURCES})
        target_link_libraries(openframe_ui_tests PRIVATE openframe_integration Qt6::Test)
        if(WIN32)
            target_link_libraries(openframe_ui_tests PRIVATE Qt6::GuiPrivate shell32 ole32)
        endif()
        if(MSVC)
            target_compile_options(openframe_ui_tests PRIVATE /utf-8 /EHsc)
        endif()
        add_test(NAME openframe_ui COMMAND openframe_ui_tests)
        set_tests_properties(openframe_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
    endif()
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/ApiTests.cpp")
        add_executable(openframe_api_tests tests/integration/ApiTests.cpp)
        target_link_libraries(openframe_api_tests PRIVATE openframe_integration Qt6::Test)
        add_test(NAME openframe_api COMMAND openframe_api_tests)
        set_tests_properties(openframe_api PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
endif()
